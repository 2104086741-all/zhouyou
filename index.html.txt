<script>
    function go(type) {
        let text = document.getElementById('kw').value.trim();
        if(!text) { alert('请输入关键词'); return; }
        let q = encodeURIComponent(text);
        let urls = {
            bili: `https://search.bilibili.com/all?keyword=${q}`,
            xhs: `https://www.xiaohongshu.com/search_result?keyword=${q}`,
            zhihu: `https://www.zhihu.com/search?type=content&q=${q}`,
            douyin: `https://www.douyin.com/search/${q}`
        };
        window.location.href = urls[type];
    }
</script>
